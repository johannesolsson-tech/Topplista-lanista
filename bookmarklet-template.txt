javascript:(async function run(c){
  try {
    if(!["lanista.se","www.lanista.se"].includes(location.hostname)) throw Error("Öppna lanista.se och logga in först.");
    const get=async path=>{const response=await fetch(path,{credentials:"include"});if(!response.ok)throw Error(path+" HTTP "+response.status);return response.json();};
    const unwrap=x=>Array.isArray(x)?x:(Array.isArray(x?.data)?x.data:Array.isArray(x?.members)?x.members:[]);
    const [top,clan]=await Promise.all([get("/api/avatars/livetoplist?type=todays_xp"),get("/api/clans/2/members?paginate=false")]);
    const members=unwrap(clan);
    const memberIds=new Set();
    const names=new Set();
    for(const m of members){
      const a=m?.avatar||m;
      const id=a?.id??m?.avatar_id;
      if(id!=null)memberIds.add(String(id));
      if(a?.name)names.add(String(a.name).trim().toLocaleLowerCase("sv-SE"));
    }
    const entries=[];
    const ranking=Array.isArray(top?.toplist)?top.toplist:(Array.isArray(top)?top:[]);
    for(const [i,p] of ranking.entries()){
      const id=p?.id??p?.avatar_id??p?.avatar?.id;
      const key=String(p?.name||"").trim().toLocaleLowerCase("sv-SE");
      if(!memberIds.has(String(id))&&!names.has(key))continue;
      entries.push({name:p.name,xp:p.todays_xp,globalRank:i+1,level:p.display_level??p.current_level??""});
    }
    const response=await fetch(c.url+"/update",{method:"POST",headers:{"Content-Type":"text/plain;charset=UTF-8"},body:JSON.stringify({secret:c.secret,entries,updatedAt:new Date().toISOString()})});
    const message=await response.text();
    if(!response.ok)throw Error("Railway HTTP "+response.status+": "+message);
    alert("Discord uppdaterat! "+entries.length+" klanmedlemmar hittades.");
  } catch(e){console.error(e);alert("Topplistefel: "+e.message);}
})({url:"DIN_RAILWAY_URL",secret:"DITT_UPDATE_SECRET"})
